cmake_minimum_required(VERSION 3.24)
project(omp-mmdb VERSION 0.1.0 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
option(MMDB_BUILD_COMPONENT "Build the open.mp x86 component" ON)
option(MMDB_BUILD_TESTS "Build and register automated tests" ON)
option(MMDB_BUILD_BENCHMARKS "Build reproducible benchmark executable" ON)
option(MMDB_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
option(MMDB_THREAD_SANITIZER "Enable ThreadSanitizer" OFF)
option(MMDB_BUILD_FUZZERS "Build libFuzzer input boundary harness" OFF)
option(MMDB_WARNINGS_AS_ERRORS "Treat omp-mmdb warnings as errors" ON)

# Older SDK transitive CMake files still declare pre-3.5 policies. This floor
# changes their CMake policy interpretation; C++ diagnostics remain enabled.
set(CMAKE_POLICY_VERSION_MINIMUM 3.5)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(MAXMINDDB_BUILD_BINARIES OFF CACHE BOOL "" FORCE)
set(MAXMINDDB_INSTALL OFF CACHE BOOL "" FORCE)
add_subdirectory(lib/libmaxminddb EXCLUDE_FROM_ALL)
# Dependency headers use extensions such as uint128. Keep diagnostics strict on
# omp-mmdb itself while treating upstream headers as external system headers.
get_target_property(MMDB_MMDB_INCLUDES maxminddb INTERFACE_INCLUDE_DIRECTORIES)
set_property(TARGET maxminddb PROPERTY INTERFACE_SYSTEM_INCLUDE_DIRECTORIES "${MMDB_MMDB_INCLUDES}")

if(MMDB_SANITIZERS AND MMDB_THREAD_SANITIZER)
	message(FATAL_ERROR "Use separate builds for ASan+UBSan and TSan")
endif()

if(MMDB_THREAD_SANITIZER)
	if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU" OR WIN32)
		message(FATAL_ERROR "TSan configuration requires Linux GCC or Clang")
	endif()
	add_compile_options(-fsanitize=thread -fno-omit-frame-pointer)
	add_link_options(-fsanitize=thread)
	target_compile_options(maxminddb PRIVATE -fsanitize=thread -fno-omit-frame-pointer)
endif()

if(MMDB_SANITIZERS)
	if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU" OR WIN32)
		message(FATAL_ERROR "ASan+UBSan configuration requires Linux GCC or Clang")
	endif()
	add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
	add_link_options(-fsanitize=address,undefined)
	target_compile_options(maxminddb PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
endif()

function(omp_mmdb_target target)
	if(MSVC)
		target_compile_options(${target} PRIVATE /W4 /permissive- /sdl /guard:cf)
		target_link_options(${target} PRIVATE /guard:cf /DYNAMICBASE /NXCOMPAT)
		if(MMDB_WARNINGS_AS_ERRORS)
			target_compile_options(${target} PRIVATE /WX)
		endif()
	else()
		target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion -Wformat=2 -fstack-protector-strong)
		if(MMDB_WARNINGS_AS_ERRORS)
			target_compile_options(${target} PRIVATE -Werror)
		endif()
		if(UNIX AND NOT APPLE)
			target_link_options(${target} PRIVATE -Wl,-z,relro,-z,now -Wl,--no-undefined)
		endif()
	endif()
endfunction()

find_package(Threads REQUIRED)
add_library(omp_mmdb_core STATIC src/core.cpp src/store.cpp)
target_include_directories(omp_mmdb_core PUBLIC src)
target_link_libraries(omp_mmdb_core PUBLIC maxminddb::maxminddb Threads::Threads)
if(WIN32)
	target_compile_definitions(omp_mmdb_core PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN)
	target_link_libraries(omp_mmdb_core PUBLIC ws2_32 psapi)
endif()
omp_mmdb_target(omp_mmdb_core)

add_library(omp_mmdb_pawn INTERFACE)
target_link_libraries(omp_mmdb_pawn INTERFACE omp_mmdb_core)
target_include_directories(omp_mmdb_pawn SYSTEM INTERFACE lib/compiler/source)
if(UNIX)
	target_include_directories(omp_mmdb_pawn SYSTEM INTERFACE lib/compiler/source/linux)
endif()
target_compile_definitions(omp_mmdb_pawn INTERFACE PAWN_CELL_SIZE=32 HAVE_STDINT_H=1)
if(WIN32)
	# MinGW uses the Windows stack allocator; it has no POSIX alloca.h.
	target_compile_definitions(omp_mmdb_pawn INTERFACE HAVE_ALLOCA_H=0)
endif()

if(MMDB_BUILD_COMPONENT)
	if(NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
		message(FATAL_ERROR "The distributed open.mp component requires x86. Configure -A Win32 or Linux -m32. Core-only x64 tests: -DMMDB_BUILD_COMPONENT=OFF")
	endif()
	add_subdirectory(lib/sdk EXCLUDE_FROM_ALL)
	get_target_property(MMDB_SDK_INCLUDES OMP-SDK INTERFACE_INCLUDE_DIRECTORIES)
	set_property(TARGET OMP-SDK PROPERTY INTERFACE_SYSTEM_INCLUDE_DIRECTORIES "${MMDB_SDK_INCLUDES}")
	add_library(omp_mmdb MODULE src/component.cpp src/pawn_bridge.cpp)
	target_link_libraries(omp_mmdb PRIVATE omp_mmdb_pawn OMP-SDK)
	set_target_properties(omp_mmdb PROPERTIES PREFIX "" OUTPUT_NAME "MMDB" CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON)
	omp_mmdb_target(omp_mmdb)
	install(TARGETS omp_mmdb LIBRARY DESTINATION components RUNTIME DESTINATION components)
endif()

set(MMDB_FIXTURE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/lib/libmaxminddb/t/maxmind-db/test-data" CACHE PATH "Redistributable MaxMind synthetic fixtures")
if(MMDB_BUILD_TESTS)
	enable_testing()
	add_executable(omp_mmdb_core_tests tests/core_tests.cpp)
	target_link_libraries(omp_mmdb_core_tests PRIVATE omp_mmdb_core)
	target_compile_definitions(omp_mmdb_core_tests PRIVATE MMDB_FIXTURE_DIR="${MMDB_FIXTURE_DIR}")
	omp_mmdb_target(omp_mmdb_core_tests)
	add_test(NAME core COMMAND omp_mmdb_core_tests)
	set_tests_properties(core PROPERTIES TIMEOUT 180)
	add_executable(omp_mmdb_store_tests tests/store_tests.cpp)
	target_link_libraries(omp_mmdb_store_tests PRIVATE omp_mmdb_core)
	target_compile_definitions(omp_mmdb_store_tests PRIVATE MMDB_FIXTURE_DIR="${MMDB_FIXTURE_DIR}")
	omp_mmdb_target(omp_mmdb_store_tests)
	add_test(NAME store COMMAND omp_mmdb_store_tests)
	set_tests_properties(store PROPERTIES TIMEOUT 180)
	add_executable(omp_mmdb_native_tests tests/native_tests.cpp src/pawn_bridge.cpp)
	target_link_libraries(omp_mmdb_native_tests PRIVATE omp_mmdb_pawn)
	target_compile_definitions(omp_mmdb_native_tests PRIVATE MMDB_FIXTURE_DIR="${MMDB_FIXTURE_DIR}")
	omp_mmdb_target(omp_mmdb_native_tests)
	add_test(NAME pawn_boundary COMMAND omp_mmdb_native_tests)
	set_tests_properties(pawn_boundary PROPERTIES TIMEOUT 180)
endif()

if(MMDB_BUILD_BENCHMARKS)
	add_executable(omp_mmdb_benchmark benchmarks/benchmark.cpp)
	target_link_libraries(omp_mmdb_benchmark PRIVATE omp_mmdb_core)
	target_compile_definitions(omp_mmdb_benchmark PRIVATE MMDB_FIXTURE_DIR="${MMDB_FIXTURE_DIR}")
	omp_mmdb_target(omp_mmdb_benchmark)
endif()

if(MMDB_BUILD_FUZZERS)
	if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang" OR WIN32)
		message(FATAL_ERROR "libFuzzer target requires Linux Clang")
	endif()
	if(NOT MMDB_SANITIZERS)
		message(FATAL_ERROR "Fuzzers require MMDB_SANITIZERS=ON to instrument the database dependency")
	endif()
	# Compile the core separately so libFuzzer observes parser/conversion branch
	# coverage, while ordinary tests continue to link a non-fuzzer core.
	add_library(omp_mmdb_fuzz_core STATIC src/core.cpp src/store.cpp)
	target_include_directories(omp_mmdb_fuzz_core PUBLIC src)
	target_link_libraries(omp_mmdb_fuzz_core PUBLIC maxminddb::maxminddb Threads::Threads)
	target_compile_options(omp_mmdb_fuzz_core PRIVATE -fsanitize=fuzzer-no-link,address,undefined)
	omp_mmdb_target(omp_mmdb_fuzz_core)
	add_executable(omp_mmdb_parser_fuzz fuzz/parser_fuzz.cpp)
	target_link_libraries(omp_mmdb_parser_fuzz PRIVATE omp_mmdb_fuzz_core)
	target_compile_options(omp_mmdb_parser_fuzz PRIVATE -fsanitize=fuzzer,address,undefined)
	target_link_options(omp_mmdb_parser_fuzz PRIVATE -fsanitize=fuzzer,address,undefined)
	add_executable(omp_mmdb_native_fuzz fuzz/native_fuzz.cpp src/pawn_bridge.cpp)
	target_link_libraries(omp_mmdb_native_fuzz PRIVATE omp_mmdb_fuzz_core)
	target_include_directories(omp_mmdb_native_fuzz SYSTEM PRIVATE lib/compiler/source lib/compiler/source/linux)
	target_compile_definitions(omp_mmdb_native_fuzz PRIVATE PAWN_CELL_SIZE=32 HAVE_STDINT_H=1 MMDB_FIXTURE_DIR="${MMDB_FIXTURE_DIR}")
	target_compile_options(omp_mmdb_native_fuzz PRIVATE -fsanitize=fuzzer,address,undefined)
	target_link_options(omp_mmdb_native_fuzz PRIVATE -fsanitize=fuzzer,address,undefined)
	omp_mmdb_target(omp_mmdb_native_fuzz)
endif()
